# Up to date as of 1.15.0
# Root = Character
on_heir_born = {
    effect = {
        if = {
            limit = {
                owner = {
                    has_variable = no_heirs
                }
            }
            kill_character = {
                value  = yes
                hidden = yes
            }
        }
        else = { # Vanilla logic sits in this else block
            if = {
                limit = {
                    NOT = { # Iesada Tokugawa cannot produce heirs
                        owner.ruler ?= {
                            has_template = JAP_iesada_tokugawa
                        }
                    }
                    owner ?= {
                        has_gov_regency = no
                    }
                }
                post_notification = heir_born_notification
            }
            else = { # Regents should not produce heirs, TODO remove this hack once we have code support
                this = { kill_character = { hidden = yes } }
            }
            if = {
                limit = {
                    owner = {
                        c:FRA ?= this
                    }
                }
                on_heir_born_french_heir_ideology_effect = yes
            }
            ### Royal House Variables
            on_heir_born_inherit_royal_house_effect = yes
        }
    }
}
