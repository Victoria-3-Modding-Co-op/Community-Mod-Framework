on_new_ruler = {
	effect = {
		if = {
			limit = {
				# CMF Addition to block regencies
				owner = {
					NOT = {
						has_variable = com_no_regencies
					}
				}
			}
			
			#### VANILLA SCRIPT BEYOND THIS POINT
			# Last updated for patch 1.13.9 on 9 AUG 2026
			if = {
				limit = {
					owner = { # Assigning a regent should never fire this on_action
						has_gov_regency = no
					}
				}
				if = {
					limit = {
						owner = {
							NOT = {
								has_variable = hide_ruler_change_notification
							}
						}
						is_adult = yes
					}
					post_notification = new_ruler
				}
				if = {
					limit = {
						owner = {
							has_law_or_variant = law_type:law_theocracy
						}
						NOT = { religion = owner.religion }
					}
					change_character_religion = owner.religion
				}

				# Setup regencies
				if = {
					limit = {
						owner = {
							has_law = law_type:law_monarchy
							has_gov_regency = no
						}
						is_adult = no
						NOT = { has_variable = regency_complete }
						NOT = { has_template = JAP_iemochi_tokugawa } # Has his own event
					}
					owner = {
						trigger_event = {
							id = regency.1
							popup = yes
						}
					}
				}

				# For handling the Sikh Maharaja achievement
				if = {
					limit = {
						owner = {
							c:PAN ?= this
							has_journal_entry = je_sikh_sovereignty
						}
					}
					on_new_ruler_sikh_maharaja_effect = yes
				}

				# For transitioning Zanzibar out of Arab rule
				if = {
					limit = {
						owner = {
							c:ZAN ?= this
							any_primary_culture = {
								cu:bedouin = this
							}
						}
						culture = {
							NOT = { cu:bedouin = this }
						}
						NOT = {
							has_global_variable = decided_future_of_zanzibar
						}
					}
					owner = {
						trigger_event = { id = zanzibar.6 popup = yes }
					}
				}

				### Monarchial succession - add all successions here.
				if = {
					limit = {
						is_ruler_of_own_country = yes
						owner ?= {
							country_has_monarchy_law = yes
							country_has_non_monarchy_law = no # To handle an execution order issue
						}
					}
					### Russian Monarchs
					if = {
						limit = {
							owner ?= {
								c:RUS ?= this
							}
						}
						character_russia_ruler_succession_chain_effect = yes
					}
					### Swedish Monarchs
					if = {
						limit = {
							owner = {
								OR = {
									c:SWE ?= this
									was_formed_from = SWE
								}
							}
						}
						character_sweden_ruler_succession_chain_effect = yes
					}
					### Austrian Monarchs
					if = {
						limit = {
							owner = {
								OR = {
									c:AUS ?= this
									was_formed_from = AUS
								}
							}
						}
						character_austria_ruler_succession_chain_effect = yes
					}
					### Serbian Monarchs
					if = {
						limit = {
							owner ?= {
								OR = {
									c:SER ?= this
									was_formed_from = SER
								}
							}
						}
						character_serbia_ruler_succession_chain_effect = yes
					}
					### Ottoman Monarchs
					if = {
						limit = {
							owner ?= {
								c:TUR ?= this
							}
						}
						character_ottoman_ruler_succession_chain_effect = yes
					}
					### French Monarchs
					if = {
						limit = {
							owner ?= {
								c:FRA ?= this
							}
						}
						character_france_ruler_succession_chain_effect = yes
					}
					### Spanish Monarchs
					if = {
						limit = {
							owner ?= {
								OR = {
									c:SPA ?= this
									was_formed_from = SPA
								}
							}
						}
						character_spain_ruler_succession_chain_effect = yes
					}
					### Portuguese Monarchs
					if = {
						limit = {
							owner ?= {
								OR = {
									c:POR ?= this
									was_formed_from = POR
								}
							}
						}
						character_portugal_ruler_succession_chain_effect = yes
					}
					### Morocco Monarchs
					if = {
						limit = {
							owner ?= {
								OR = {
									c:MOR ?= this
									was_formed_from = MOR
								}
							}
						}
						character_morocco_ruler_succession_chain_effect = yes
					}
					if = {
						limit = {
							owner ?= {
								c:JAP ?= this
							}
						}
						character_japan_ruler_succession_chain_effect = yes
					}
				}

				if = { # For handling the Succession Rights treaty article in the case of ruler death
					limit = {
						exists = c:BRZ
						exists = c:POR
						owner = {
							c:POR ?= this
							NOT = {
								has_diplomatic_pact = {
									who = c:BRZ
									type = personal_union
								}
							}
							any_scope_treaty = {
								binds = c:BRZ
								any_scope_article = {
									has_type = amend_succession
									source_country = c:BRZ
								}
							}
						}
					}
					# For handling the Succession Rights treaty article in the case of ruler death
					on_new_ruler_brazil_succession_rights_effect = yes
				}

				# For handling the assignment of emperor_var, shogun_var, and emperor traits
				if = {
					limit = {
						c:JAP ?= owner
					}
					on_new_ruler_japan_effect = yes
				}
			}

			if = { # Handles total years in power, not consecutive years. Mostly for Japan.
				limit = {
					NOT = {
						has_variable = year_in_power
					}
				}
				set_variable = { name = year_in_power value = 1 } # In case all else fails
			}
		}
	}
}
