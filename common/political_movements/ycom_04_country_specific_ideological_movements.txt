REPLACE:movement_utilitarian = {
	category = movement_category_ideological
	
	active_law_radicalism_multiplier = 0.5

	ideology = ideology_utilitarian_movement
	character_ideologies = {
		ideology_utilitarian_leader
	}

    disband_trigger = {
        has_council_republic_law = yes
    }
    on_disbanded = {
        fold_movement_into_anti_communist_movement_effect = yes
    }

	creation_trigger = {
		OR = {
			country_has_primary_culture = cu:british
			country_has_primary_culture = cu:anglo_canadian
			country_has_primary_culture = cu:australian
			anz_country_is_pakeha = yes #anzfp
		}
	}
	creation_weight = {
		value = 100
		if = {
			limit = {
				owner ?= { # If a movement of this type exists in a neighbouring country, make it more likely to spawn here as well
					any_neighbouring_state = {
						owner = {
							prestige > ROOT.owner.prestige
							any_political_movement = {
								is_political_movement_type = movement_utilitarian
							}
						}
					}
				}
			}
			multiply = {
				value = 2
			}
		}
	}
	on_created = {
		add_movement_enthusiasm_modifier = yes
	}
	
	character_support_trigger = {
		OR = {
			AND = {
				culture = {
					has_discrimination_trait_group = heritage_group_european
				}
				has_reactionary_ideology = no
				has_conservative_ideology = no
				has_socialist_ideology = no
			}
			has_ideology = ideology:ideology_utilitarian_leader
		}
	}
	character_support_weight = {
		value = 200
		if = {
			limit = {
				has_ideology = ideology:ideology_utilitarian_leader
			}
			multiply = {
				value = 5
			}
		}
	}
	
	pop_support_trigger = {
		OR = {
			AND = {
				owner = {
					country_is_colonial_or_company = yes
				}
				culture = {
					has_discrimination_trait_group = heritage_group_european
				}
			}
			is_pop_type = bureaucrats
			is_pop_type = capitalists
			is_pop_type = engineers
			is_pop_type = clerks
			is_pop_type = officers
			is_pop_type = academics
		}
		NOR = {
			is_pop_type = laborers
			is_pop_type = peasants
			is_pop_type = slaves
		}
		mog_plp_movement_mainland_general_pop_support_trigger = yes    # Grey_PLP
	}
	
	pop_support_factors = {
		movement_support_colonial_rule
		movement_support_high_acceptance
		movement_support_bureaucrats
		movement_support_academics
		movement_support_officers
		movement_support_clerks
	}
	
	pop_support_weight = {
		if = {
			limit = {
				exists = scope:political_movement
				scope:political_movement ?= {
					OR = {
						is_being_suppressed = yes
						is_being_bolstered = yes
					}
				}
			}
			if = {
				limit = {
					scope:political_movement = {
						is_being_suppressed = yes
					}
				}
				add = {
					value = suppression_base_value
					multiply = {
						value = -0.5
						add = owner.modifier:country_suppression_attraction_factor
					}
					desc = "concept_suppressed"
				}
			}
			else = {
				add = {
					value = bolstering_base_value
					multiply = {
						value = 0.5
						add = owner.modifier:country_bolster_attraction_factor
					}
					desc = "concept_bolstered"
				}
			}
		}
		
		add = {
			value = 1
			desc = "POP_BASE_SUPPORT"
		}
		
		if = {
			limit = {
				owner = {
					country_is_colonial_or_company = yes
				}
				culture = {
					has_discrimination_trait_group = heritage_group_european
				}
			}
			add = {
				desc = "colonial_rule"
				value = 15
				if = {
					limit = {
						is_pop_type = bureaucrats
					}
					multiply = 1.5
				}
			}
		}
		
		if = {
			limit = {
				is_pop_type = bureaucrats
			}
			add = {
				desc = "POP_BUREAUCRATS"
				value = 9
			}
		}
		else_if = {
			limit = {
				is_pop_type = academics
			}
			add = {
				desc = "POP_ACADEMICS"
				value = 6
			}
		}
		else_if = {
			limit = {
				is_pop_type = capitalists
			}
			add = {
				desc = "POP_CAPITALISTS"
				value = 3
			}
		}
		else_if = {
			limit = {
				is_pop_type = officers
			}
			add = {
				desc = "POP_OFFICERS"
				value = 3
			}
		}
		else_if = {
			limit = {
				is_pop_type = clerks
			}
			add = {
				desc = "POP_CLERKS"
				value = 3
			}
		}
		
		multiply = { # Illiterate pops participate at 1/10 the rate of fully literate pops
			desc = "POP_LITERACY"
			value = literacy_rate
			add = 0.10
		}
		
		multiply = {
			desc = "POP_ACCEPTANCE"
			value = pop_acceptance
			divide = 75
		}
		
		# Grey_PLP - always put last
		multiply = {
			value = mog_plp_movement_mainland_general_pop_support_weight_incorporation_percent_mult
		}
	}
	
	revolution = {
		possible = {
			political_movement_support >= 0.05
		}
		
		weight = {
			value = 100
		}
		
		state_weight = {
			value = 1
			if = {
				limit = {
					OR = {
						any_scope_building = {
							is_building_group = bg_bureaucracy
							level >= 10
						}
					}
				}
				multiply = 10
			}
			else_if = {
				limit = {
					OR = {
						any_scope_building = {
							is_building_group = bg_bureaucracy
							level >= 5
						}
					}
				}
				multiply = 5
			}
			else = {
				multiply = 0
			}
			# Grey_PLP - always put last
			multiply = {
				value = mog_plp_movement_all_general_state_weight_mult
			}
		}
	}
	
	secession = {
		possible = {
			political_movement_support >= 0.05
			scope:clout <= 0
		}
		
		weight = {
			value = 1000
		}
		
		state_weight = {
			value = 1
			if = {
				limit = {
					OR = {
						any_scope_building = {
							is_building_group = bg_bureaucracy
							level >= 10
						}
					}
				}
				multiply = 10
			}
			else_if = {
				limit = {
					OR = {
						any_scope_building = {
							is_building_group = bg_bureaucracy
							level >= 5
						}
					}
				}
				multiply = 5
			}
			else = {
				multiply = 0
			}
			# Grey_PLP - always put last
			multiply = {
				value = mog_plp_movement_all_general_state_weight_mult
			}
		}
	}
	
	additional_radicalism_factors = {
		add = movement_radicalism_schools_funding
		if = {
			limit = {
				neighboring_country_has_insurrectionary_movement_trigger = { TYPE = movement_utilitarian }
			}
			add = {
				desc = "NEIGHBOURING_COUNTRY_WITH_REVOLUTION"
				value = 0.15
			}
		}
		add = movement_radicalism_suppressed_after_civil_war
	}
}

			if = {
				limit = {
					scope:political_movement = {
						is_being_suppressed = yes
					}
				}
				add = {
					value = suppression_base_value
					multiply = {
						value = -0.5
						add = owner.modifier:country_suppression_attraction_factor
					}
					desc = "concept_suppressed"
				}
			}
			else = {
				add = {
					value = bolstering_base_value
					multiply = {
						value = 0.5
						add = owner.modifier:country_bolster_attraction_factor
					}
					desc = "concept_bolstered"
				}
			}
		}

		if = { # Cossacks at home
			limit = {
				pop_employment_building = building_cossack_host
			}
			add = {
				value = 30
				desc = "POP_SUPPORT_COSSACKS"
			}
		}
		else_if = {
			limit = {
				state = {
					has_recently_removed_cossack_host = yes
				}
				OR = {
					is_pop_type = farmers
					is_pop_type = laborers
				}
			}
			add = {
				value = 10
				desc = "POP_SUPPORT_COSSACKS_REMOVED"
			}
		}
		else_if = { # Cossacks at the front
			limit = {
				OR = {
					pop_employment_building = building_barrack
					pop_employment_building = building_conscription_center
				}
			}
			add = {
				value = 15
				desc = "POP_SUPPORT_COSSACKS_MILITARY"
			}
		}
		else = { # Everyone else in the neighborhood
			add = {
				value = 1
				desc = "POP_SUPPORT_COSSACKS_PRESENT"
			}

			if = {
				limit = {
					strata = lower
				}

				add = {
					value = 2
					desc = "LOWER_NO_ICON"
				}
			}
		}

		if = {
			limit = {
				owner ?= {
					NOT = {
						any_primary_culture = {
							OR = {
								has_discrimination_trait = heritage_east_slavic
								has_discrimination_trait = tradition_cossacks
							}
						}
					}
				}
			}
			multiply = {
				value = 1.5
				desc = "COSSACKS_UNDER_NON_COSSACK_COUNTRY"
			}
		}
		
		# Grey_PLP - always put last
		multiply = {
			value = mog_plp_movement_mainland_general_pop_support_weight_incorporation_percent_mult
		}
	}

	revolution = {
		possible = {
			political_movement_support >= 0.05
		}

		weight = {
			value = 100
		}

		state_weight = {
			value = 0

			if = {
				limit = {
					OR = {
						has_building = building_cossack_host
						has_recently_removed_cossack_host = yes
					}
				}
				add = scope:clout
			}
			
			# Grey_PLP
			multiply = {
				value = mog_plp_movement_all_general_state_weight_mult
			}
		}
	}

	secession = {
		possible = {
			OR = {
				AND = {
					political_movement_support >= 0.05
					scope:clout <= 0
				}
				owner = {
					NOT = {
						any_primary_culture = {
							OR = {
								has_discrimination_trait = heritage_east_slavic
								has_discrimination_trait = tradition_cossacks
							}
						}
					}
				}
			}
		}

		weight = {
			value = 1000
		}

		state_weight = {
			value = 0

			if = {
				limit = {
					OR = {
						has_building = building_cossack_host
						has_recently_removed_cossack_host = yes
					}
				}
				add = 0.5
			}
			
			# Grey_PLP
			multiply = {
				value = mog_plp_movement_all_general_state_weight_mult
			}
		}
	}

	additional_radicalism_factors = {
		if = {
			limit = {
				owner ?= {
					NOT = {
						any_primary_culture = {
							OR = {
								has_discrimination_trait = heritage_east_slavic
								has_discrimination_trait = tradition_cossacks
							}
						}
					}
				}
			}
			add = {
				value = 0.15
				desc = "COSSACKS_UNDER_NON_COSSACK_COUNTRY"
			}
		}

		if = {
			limit = {
				owner ?= {
					any_scope_state = {
						has_recently_removed_cossack_host = yes
					}
				}
			}
			owner ?= {
				every_scope_state = {
					limit = {
						has_recently_removed_cossack_host = yes
					}
					add = {
						value = 0.1
						desc = "RECENTLY_REMOVED_COSSACKS_HOST"
					}
				}
			}
		}

		if = {
			limit = {
				neighboring_country_has_insurrectionary_movement_trigger = { TYPE = movement_cossack }
			}
			add = {
				desc = "NEIGHBOURING_COUNTRY_WITH_REVOLUTION"
				value = 0.15
			}
		}
		add = movement_radicalism_suppressed_after_civil_war
	}
}
