REPLACE_OR_CREATE:com_change_global_variable = {
    set_global_variable = {
        name = $name$
        value = {
            if = {
                limit = {
                    has_global_variable = $name$
                }
                add = global_var:$name$
            }
            add = $value$
        }
    }
}

REPLACE_OR_CREATE:com_update_value_in_global_variable_map = {
    if = {
        limit = {
            is_key_in_global_variable_map = {
                name = $name$
                target = $key$
            }
        }
        remove_from_global_variable_map = {
            name = $name$
            key = $key$
        }
    }
    add_to_global_variable_map = {
        name = $name$
        key = $key$
        value = $value$
    }
}

####---------- CMF Variable / Flag Error Suppression Effect ----------####
REPLACE_OR_CREATE:fix_variable_error = {
    if = {
        limit = {
            has_variable = $variable$
            var:$variable$ = flag:$variable$
            exists = scope:$variable$
            scope:$variable$ = flag:$variable$
        }
        set_variable = {
            name = $variable$
            value = flag:$variable$
        }
        flag:$variable$ = {
            save_scope_as = $variable$
        }
    }
}

####---------- Mod Debug Menu Manager Effects ----------####
com_add_mod_to_debug_manager_effect = {
    create_container = {
        name = $mod_name$
        tags = { mod_debug }
        save_scope_as = com_mod_container_scope
        on_created = {
            set_variable = {
                name = com_mod_name
                value = flag:$mod_name$
            }
            set_variable = {
                name = com_mod_command
                value = flag:$mod_command$
            }
        }
    }
    add_to_global_variable_list = {
        name = com_mod_debug_manager_list
        target = scope:com_mod_container_scope
    }
}

####---------- Custom Error Deer Icon Effects ----------####
# NOTE: I will refactor this to add to a global list; functionality will be unchanged from the user perspective -Alexedishi
com_add_custom_error_deer_icon = {
    if = {
        limit = {
            NOT = {
                is_target_in_global_variable_list = {
                    name = com_error_deer_icon_list
                    target = ig_trait:$icon$
                }
            }
        }
        add_to_global_variable_list = {
            name = com_error_deer_icon_list
            target = ig_trait:$icon$
        }
    }
    set_global_variable = {
        name = com_error_deer_icon_var
        value = ig_trait:$icon$
    }
}

com_set_custom_error_deer_icon = {
    if = {
        limit = {
            is_target_in_global_variable_list = {
                name = com_error_deer_icon_list
                target = ig_trait:$icon$
            }
        }
        set_global_variable = {
            name = com_error_deer_icon_var
            value = ig_trait:$icon$
        }
    }
    else = {
        error_log = "Provided ig_trait was not found in global list com_error_deer_icon_list"
    }
}
com_reset_custom_error_deer_icon = {
    remove_global_variable = com_error_deer_icon_var
}
