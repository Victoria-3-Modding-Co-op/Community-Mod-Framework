com_situation_start_date_today_effect = {
	if = {
		limit = {
			month = 0
		}
		set_variable = {
			name = com_situation_start_month_var
			value = flag:JANUARY
		}
	}
	else_if = {
		limit = {
			month = 1
		}
		set_variable = {
			name = com_situation_start_month_var
			value = flag:FEBRUARY
		}
	}
	else_if = {
		limit = {
			month = 2
		}
		set_variable = {
			name = com_situation_start_month_var
			value = flag:MARCH
		}
	}
	else_if = {
		limit = {
			month = 3
		}
		set_variable = {
			name = com_situation_start_month_var
			value = flag:APRIL
		}
	}
	else_if = {
		limit = {
			month = 4
		}
		set_variable = {
			name = com_situation_start_month_var
			value = flag:MAY
		}
	}
	else_if = {
		limit = {
			month = 5
		}
		set_variable = {
			name = com_situation_start_month_var
			value = flag:JUNE
		}
	}
	else_if = {
		limit = {
			month = 6
		}
		set_variable = {
			name = com_situation_start_month_var
			value = flag:JULY
		}
	}
	else_if = {
		limit = {
			month = 7
		}
		set_variable = {
			name = com_situation_start_month_var
			value = flag:AUGUST
		}
	}
	else_if = {
		limit = {
			month = 8
		}
		set_variable = {
			name = com_situation_start_month_var
			value = flag:SEPTEMBER
		}
	}
	else_if = {
		limit = {
			month = 9
		}
		set_variable = {
			name = com_situation_start_month_var
			value = flag:OCTOBER
		}
	}
	else_if = {
		limit = {
			month = 10
		}
		set_variable = {
			name = com_situation_start_month_var
			value = flag:NOVEMBER
		}
	}
	else_if = {
		limit = {
			month = 11
		}
		set_variable = {
			name = com_situation_start_month_var
			value = flag:DECEMBER
		}
	}
	else = {
		error_log = "Effect com_situation_start_date_today_effect was called but the month value was out of range. How did this happen?"
	}
	set_variable = {
		name = com_situation_start_year_var
		value = year
	}
}

	}
}

####---------- Situation Phase Effects ----------####
com_set_situation_phase_effect = {
	set_variable = {
		name = com_situation_phase_icon_var
		value = flag:$phase_icon$
	}
	set_variable = {
		name = com_situation_phase_name_var
		value = flag:$phase_name$
	}
}

com_set_indexed_situation_phase_progress_details_effect = {
	save_scope_value_as = {
		name = com_index
		value = $index$
	}
	if = {
		limit = {
			OR = {
				scope:com_index = 1
				scope:com_index = 2
				scope:com_index = 3
				scope:com_index = 4
			}
		}
		set_variable = {
			name = com_situation_group_$index$_next_phase_name_var
			value = flag:$next_phase_name$
		}
		set_variable = {
			name = com_situation_group_$index$_next_phase_icon_var
			value = flag:$next_phase_icon$
		}
		set_variable = {
			name = com_situation_group_$index$_next_phase_progress_var
			value = $progress_value$
		}
		set_variable = {
			name = com_situation_group_$index$_next_phase_max_progress_var
			value = $max_value$
		}
		set_variable = {
			name = com_situation_group_$index$_next_phase_min_progress_var
			value = $min_value$
		}
		set_variable = {
			name = com_situation_group_$index$_next_phase_factors_var
			value = flag:$factors$
		}
	}
	else = {
		error_log = "Effect com_set_indexed_situation_phase_progress_details_effect requires an index value of 1, 2, 3, or 4."
	}
}

com_hide_indexed_situation_phase_progress_details_effect = {
	# WARNING this effect assumes you have already set all other variables via com_set_indexed_situation_phase_progress_details_effect first!
	# This only removes com_situation_group_$index$_next_phase_progress_var, which is used by the UI to hide and show the progress element for that index number
	# This effect can be reversed by running any of the other effects that update com_situation_group_$index$_next_phase_progress_var
	save_scope_value_as = {
		name = com_index
		value = $index$
	}
	if = {
		limit = {
			OR = {
				scope:com_index = 1
				scope:com_index = 2
				scope:com_index = 3
				scope:com_index = 4
			}
		}
		remove_variable = com_situation_group_$index$_next_phase_progress_var
	}
	else = {
		error_log = "Effect com_hide_indexed_situation_phase_progress_details_effect requires an index value of 1, 2, 3, or 4."
	}
}

####---------- Situation Outcome Effects ----------####
com_create_indexed_situation_outcome_details_effect = {
	save_scope_value_as = {
		name = com_index
		value = $index$
	}
	if = {
		limit = {
			OR = {
				scope:com_index = 0
				scope:com_index = 1
				scope:com_index = 2
			}
		}
		set_variable = {
			name = com_situation_outcome_$index$_name_var
			value = flag:$name$
		}
		set_variable = {
			name = com_situation_outcome_$index$_illustration_var
			value = ig_trait:$texture$
		}
	}
	else = {
		error_log = "Effect com_create_indexed_situation_outcome_details_effect requires an index value of 0, 1, or 2. This must match the index of the journal outcome, starting from 0."
	}
}
