####---------- Progress Bar Styling Effects ----------####
# Documentation: https://github.com/Victoria-3-Modding-Co-op/Community-Mod-Framework/wiki/Stylized-Journal-Progress-Bars

# Check whether the given progress bar id exists
# Parameters:
# - progress_bar = progress bar localization key
REPLACE_OR_CREATE:exists_com_progress_bar = {
    custom_description = {
        text = exists_com_progress_bar
        container_exists = $progress_bar$
    }
}

# Check whether the given progress bar has been hidden
# Parameters:
# - progress_bar = progress bar localization key
REPLACE_OR_CREATE:is_com_progress_bar_hidden = {
    custom_description = {
        text = is_com_progress_bar_hidden
        container:$progress_bar$ = {
            has_variable = com_hide_progress_bar
        }
    }
}
